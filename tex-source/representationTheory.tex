\metatitle{Representation theory}
\metadescription{An introduction to representation theory, including general definitions, irreducible representations,
tensor products, characters, and specific focus on the representation theory
of the symmetric group $\symS_n$ and the general linear group $\GL_n$.}

\section[representationTheory]{Representation theory}

\subsection[representationDefinition]{Representation of a group}


Let $G$ be a group and $V$ a vector space. A
\defin{representation}\label{groupRepresentation}
is a map $\rho$ that sends elements in $G$ to linear maps on $V$,
with the condition that it is compatible with the group structure.
To be precise, let $\psi_g : V \to V$ be the linear map on $V$ we associate
with $g \in G$.
Then
\[
\psi_{g_1 g_2} = \psi_{g_1} \circ \psi_{g_2} \text{ for all } g_1,g_2 \in G.
\]
Thus, a representation is a group homomorphism $\rho : G \to \GL(V)$.

\begin{example}
Let $G = \symS_n$ and $V=\setC[x_1,\dotsc,x_n]$.
Note that $V$ is not finite-dimensional.
Then we let $G$ act on $V$ by permuting the indices of the variables.
After choosing a monomial basis of $V$, the map $\rho$ sends a permutation
$g \in G$ to a permutation matrix in $\GL(V)$.
\end{example}


\subsection[representationIrreducible]{Irreducible representation}

Let $W \subset V$ be a subspace and let $\rho:G \to \GL(V)$ be a representation.
If $\rho(g)w \in W$ for all $w \in W$, then we say that $W$ is $G$-invariant.

If the only $G$-invariant subspaces of $V$ are $V$ itself and $\{0\}$, then
we say that $\rho$ is an
\defin{irreducible representation}\label{irreducibleRepresentationDefinition}.

For a finite group $G$ over $\setC$, every finite-dimensional representation
can be decomposed as a direct sum of irreducible representations.  That is, if
$\rho: G \to \GL(V)$ is such a representation, we can write
$V = V_1 \oplus \dotsb \oplus V_k$ such that for every $i$ we have
\[
\rho(g) v \in V_i \; \text{ for all }\; v \in V_i
\]
and the restricted representation on $V_i$ is irreducible.


\subsection[representationTensorProduct]{Tensor products}

If $\rho_1$ and $\rho_2$ are two representations of $G$, sending group
elements to elements in $\GL(V)$ and $\GL(W)$, then we define the
\defin{tensor product of representations}\label{tensorProductRepresentationDefinition}
as the map
sending $g$ to the tensor product of the individual images:
\[
(\rho_1 \otimes \rho_2)(g) \coloneqq (\rho_1(g) \otimes \rho_2(g))
\]

\subsection[representationCharacter]{Character}

Let $\rho: G \to \GL(V)$ be a representation of $G$.
Then the \defin{character}\label{representationCharacterDefinition} is the
map $\chi_{\rho}:G \to \setC$, defined as
$\chi_{\rho}(g) \coloneqq \operatorname{tr}(\rho(g))$.

A character is irreducible if it is the character of an
\hyperref[representationIrreducible]{irreducible representation}.


\section[SnRepresentationTheory]{Representation theory of $\symS_n$}

It is very common to consider $\symS_n$ acting on some vector space.
This is then an $\symS_n$-module, and a representation of $\symS_n$.

As a typical example of $\symS_n$-modules, we have the
\hyperref[spechtModules]{Specht modules}.


\subsection[frobeniusCharacteristic]{Frobenius characteristic}

Let $\chi$ be a class function on $\symS_n$.
We then define the \defin{Frobenius characteristic}
(or \defin{Frobenius image}\label{frobeniusImage})
as the map
\[
\frobChar(\chi) \coloneqq \frac{1}{n!} \sum_{\sigma \in \symS_n} \chi(\sigma) \powerSum_\sigma(\xvec)
= \sum_{\mu} \chi_\mu \frac{\powerSum_\mu(\xvec)}{z_\mu}.
\]
This map has the property that for the irreducible characters $\chi^\lambda$ of $\symS_n$,
which are class functions, we have
\[
\frobChar(\chi^\lambda) = \schurS_\lambda(\xvec).
\]

Note that the \defin{Hilbert series}\label{hilbertSeries}
can be recovered from the Frobenius characteristic.
If $M = \bigoplus_k M^k$ is a graded $\symS_n$-module, and
\[
\frobChar(M) =
\sum_{k \geq 0} t^k
\sum_{\lambda \vdash n} a(\lambda,k) \schurS_\lambda(\xvec),
\]
then the Hilbert series is given by
\[
\sum_{k \geq 0} t^k \sum_{\lambda \vdash n} a(\lambda,k) f^{\lambda},
\]
where $f^\lambda$ is the number of
\hyperref[prelimTableaux]{standard Young tableaux} of shape $\lambda$.
The Hilbert series can alternatively be described as
$\langle \frobChar(M), \completeH_{1^n} \rangle$.

\name{Lorenzo Vanzi} studies the irreducible constituents of the Sylow
permutation character of $\symS_n$ at the prime $2$ \cite{Vanzi2026x}.
If $n=2^{a_1}+\dotsb+2^{a_t}$ with $a_1>\dotsb>a_t\geq 0$ and
$n\geq 11$, then every non-hook partition of $n$ with at most
$a_1+t-1$ parts labels such a constituent, and this bound is sharp.  He also
shows that no constituent has more than $\lceil n/2\rceil$ parts.

\name{Sarah Brauner}, \name{Patricia Commins}, and \name{Victor Reiner}
study invariant theory for the free left-regular band on injective words and
for a $q$-analogue with finite general linear group symmetry
\cite{BraunerComminsReiner2023}.  Their decompositions use Stirling and
$q$-Stirling numbers, random-to-top shuffling, and derangement symmetric
functions of \name{D{\'e}sarm{\'e}nien} and \name{Wachs}.

\name{Marino Romero} decomposes Kronecker powers of the harmonics of
$\symS_n$ by studying the action of $\symS_n$ on polynomial rings with several
sets of variables \cite{Romero2022KroneckerPowers}.  The combinatorial side
uses a generalized comajor statistic, giving generalized principal evaluations
for Schur and fundamental quasisymmetric functions.

\name{Anatoly M. Vershik} and \name{Natalia V. Tsilevich} study the
\defin{Schur--Weyl graph}\label{schurWeylGraph},
the branching graph attached to Schur--Weyl
duality \cite{VershikTsilevich2021}.  Their graph-theoretic approach gives
another route to \name{Elmar Thoma}'s theorem on characters of the infinite
symmetric group.

\subsection[computingSnCharacters]{Computing characters of $\symS_n$}

Suppose we have an $\symS_n$-module $M$ and wish to compute its
\hyperref[frobeniusCharacteristic]{Frobenius characteristic}
$\frobChar_M(\xvec)$.
This gives a step-by-step guide (inspired by lecture notes by \name{Mike Zabrocki}).

The same method can be adapted to compute characters for arbitrary finite groups.

\begin{enumerate}
\item
We assume that $M$ is the span of some set of polynomials in $\setC[x_1,\dotsc,x_n]$.

\item
Use linear algebra and find a basis for $M = \langle \bvec_1,\bvec_2,\dotsc,\bvec_d \rangle$.

\item
For each $\sigma \in \symS_n$, we compute its character $\chi_M(\sigma)$ as follows.
For each $i \in [d]$, solve the linear system
 \[
 \sigma(\bvec_i) = c_{i1} \bvec_1 + c_{i2} \bvec_2 + \dotsb +  c_{id} \bvec_d.
 \]
 Then $\chi_M(\sigma) \coloneqq c_{11} + c_{22} + \dotsb + c_{dd}$.
 This is a type of trace and independent of the choice of basis.

 It is in fact sufficient to compute $\chi_M(\sigma)$ for each type $\mu$,
 as $\chi_M(\sigma)=\chi_M(\tau)$ whenever $\sigma$ and $\tau$ have the same cycle type.


\item

Now that we have computed $\chi_M(\sigma)$, we use the above formula and get that
\[
\frobChar_M(\xvec) = \frac{1}{n!} \sum_{\sigma \in \symS_n} \chi_M(\sigma) \powerSum_\sigma(\xvec).
\]

Using the fact that $\frac{n!}{z_\mu}$ is the number of permutations of type $\mu$
and the above observations, we see that
\[
\frobChar_M(\xvec) = \sum_{\mu \vdash n} \chi_M(\mu) \frac{\powerSum_\mu(\xvec)}{z_\mu},
\]
where $\chi_M(\mu)$ is the character of some permutation of type $\mu$.

\item

We then have the following relation:
 \[
  M = \bigoplus_{\lambda \vdash n} \left(S^{\lambda} \right)^{\oplus c_\lambda} \qquad
  \Longleftrightarrow
  \qquad
  \frobChar_M(\xvec) = \sum_{\lambda \vdash n} c_\lambda \schurS_\lambda(\xvec).
 \]
 Here, we have expressed $M$ as a sum of irreducible $\symS_n$-modules, denoted $S^{\lambda}$.
 In particular, we must have that the coefficients $c_\lambda$ are non-negative integers.
 Note that $\dim(S^{\lambda}) = f^\lambda$, the number of
 \hyperref[prelimTableaux]{standard Young tableaux} of shape $\lambda$.

\end{enumerate}

In the following polynomial examples, $\symS_n$ acts by permuting variable indices.

\begin{example}
Let $M = \langle x_1 + x_2 + x_3 \rangle$. This is a one-dimensional vector space,
and all elements in $\symS_3$ fix $M$.
Thus, $\chi_M(\sigma)=1$ for all $\sigma \in \symS_3$ and
\[
\frobChar_M(\xvec)
=
\frac{1}{3!} \sum_{\sigma \in \symS_3} \powerSum_\sigma(\xvec)
= \schurS_{3}(\xvec).
\]
The same thing happens for $M = \langle x_1 x_2 x_3 \rangle$.
\end{example}

\begin{example}
Let $M = \langle (x_1-x_2)(x_1-x_3)(x_2-x_3) \rangle$.
This is also a one-dimensional vector space,
but odd permutations change the sign of the basis vector.
Thus, $\chi_M(\sigma)=\sign(\sigma)$ and
\[
\frobChar_M(\xvec)
=
\frac{1}{3!}
\sum_{\sigma \in \symS_3} \sign(\sigma) \powerSum_\sigma(\xvec)
= \schurS_{111}(\xvec).
\]
\end{example}

\begin{example}
Let $M = \langle x_1 , x_2 , x_3 \rangle$.
Note that $\langle x_1 + x_2 + x_3 \rangle$ is an invariant subspace.

Then $\frobChar_M(\xvec) = \schurS_{3}(\xvec) + \schurS_{21}(\xvec)$.
\end{example}


\begin{example}

\subsection[regularMatrixModules]{Examples from regular zero--one matrices}

For $n\geq1$ and $0\leq k\leq n$, let $\mathcal M_{n,k}$ be the set
of $n\times n$ zero--one matrices with every row and column sum equal to $k$.
Independent row and column permutations give the action
$(\sigma,\tau)A=P_\sigma A P_\tau^{-1}$ of
$\symS_n\times\symS_n$, where $P_\sigma$ is a permutation matrix.
We use the \defin{formal permutation module}
$V_{n,k}=\setC[\mathcal M_{n,k}]$: its basis consists of independent symbols
$[A]$, not the numerical matrices in their ordinary linear span.
Its character counts matrices fixed by $(\sigma,\tau)$.
The examples below use the orbit--stabilizer description of permutation
modules and the Frobenius characteristic of
\hyperref[spechtTabloids]{Young permutation modules}; see \cite{Sagan2001}.

For $k\geq1$, these matrices are precisely sums of $k$ permutation matrices
with pairwise disjoint supports. Indeed, their bipartite graphs are
$k$-regular. For a set $S$ of row vertices, counting incident edges gives
$k|S|\leq k|N(S)|$, so Hall's theorem supplies a perfect matching;
removing it and inducting gives the decomposition.
Listing the column selected in each row by each matching gives a
$k\times n$ \defin{Latin rectangle}: each row is a permutation of $[n]$,
and each column has distinct entries. Thus ordered decompositions, over all
$A\in\mathcal M_{n,k}$, correspond to Latin rectangles
\cite[Sec.~1]{Wanless1999}.

\begin{example}[Row permutations and plethysm]
\label{regularMatrixRowAction}
Keep the columns labeled, and let
$R_{n,k}(X)$ be the Frobenius characteristic of the row action.
A row orbit is determined by the multiplicities $m_S$ of its row supports,
where $S$ ranges over the $k$-subsets of $[n]$. The constraints are
\[
 \sum_Sm_S=n,\qquad \sum_{S\ni j}m_S=k\quad(1\leq j\leq n).
\]
The stabilizer is $\prod_S\symS_{m_S}$, so the orbit contributes
$\prod_S\completeH_{m_S}(X)$. Consequently $R_{n,k}$ is
$h$-positive, and
\[
 R_{n,k}(X)
 =[z_1^k\dotsm z_n^k]\,
 \completeH_n[X\,\elementaryE_k(z_1,\dotsc,z_n)].
\]
Here brackets inside a symmetric function denote
\hyperref[plethysm]{plethystic substitution}; $[z_1^k\dotsm z_n^k]$
extracts an ordinary monomial coefficient.
To see the formula, expand
\[
 \sum_{\ell\geq0}\completeH_\ell[X\,\elementaryE_k(z)]t^\ell
 =\prod_{|S|=k}\left(\sum_{r\geq0}
       \completeH_r(X)t^r\prod_{j\in S}z_j^r\right),
\]
with the number of column variables fixed.
Each factor chooses one support multiplicity.
The \hyperref[schurCauchyFormula]{Cauchy identity} also gives
\[
 R_{n,k}(X)=\sum_{\lambda\vdash n}
 \left\langle\schurS_\lambda[\elementaryE_k],
                   \completeH_{(k^n)}\right\rangle\schurS_\lambda(X),
\]
where the bracket is the \hyperref[hallInnerProduct]{Hall inner product}.
For example, $R_{n,1}=\completeH_1^n$ and $R_{n,n}=\completeH_n$.
Complementation $A\mapsto J_n-A$, with $J_n$ the all-ones matrix,
gives $R_{n,k}=R_{n,n-k}$.

If entries may be arbitrary nonnegative integers, the row supports become
multisets. Writing $\widehat R_{n,k}$ for the resulting row characteristic,
the same argument replaces $\elementaryE_k$ by $\completeH_k$:
\[
 \widehat R_{n,k}(X)
 =[z_1^k\dotsm z_n^k]\,
 \completeH_n[X\,\completeH_k(z_1,\dotsc,z_n)].
\]
\end{example}

\begin{example}[Two matchings and dihedral stabilizers]
\label{regularMatrixTwoMatchings}
For $k=2$, every graph is a disjoint union of even cycles.
The row/column orbits are therefore indexed by
$\lambda=(2^{a_2}3^{a_3}\dotsm)\vdash n$, with no parts equal to $1$:
a part $m$ represents a cycle $C_{2m}$.
The color-class-preserving automorphism group of $C_{2m}$ is the
dihedral group $D_m$ of order $2m$; for $m=2$ it is
$D_2\cong\symS_2\times\symS_2$.
Thus the stabilizer of type $\lambda$ is the naturally embedded subgroup
\begin{align*}
 H_\lambda&=\prod_{m\geq2}(D_m\wr\symS_{a_m}),\\
 V_{n,2}&\cong\bigoplus_{\substack{\lambda\vdash n\\a_1=0}}
       \operatorname{Ind}_{H_\lambda}^{\symS_n\times\symS_n}\mathbf1.
\end{align*}
Here the wreath product permutes equal-sized components as well as acting
within each one. With $z_\lambda=\prod_m m^{a_m}a_m!$, the orbit has size
$(n!)^2/(2^{\ell(\lambda)}z_\lambda)$.

For the row action alone, identical rows occur exactly in the $C_4$
components. Their stabilizer is
$(\symS_2)^{a_2}\times(\symS_1)^{n-2a_2}$. Dividing the full orbit size
by the row-orbit size gives
\[
 R_{n,2}(X)=
 \sum_{\substack{\lambda\vdash n\\a_1=0}}
 \frac{n!\,2^{a_2}}{2^{\ell(\lambda)}z_\lambda}
 \completeH_2(X)^{a_2}\completeH_1(X)^{n-2a_2}.
\]
At $n=4$, the types $(4)$ and $(2,2)$ have respectively $72$ and $18$
matrices, and each type has three row orbits. Hence
\[
 R_{4,2}=3\completeH_1^4+3\completeH_2^2,\qquad
 \dim V_{4,2}=3\cdot24+3\cdot6=90.
\]

The matrices and their decompositions are different sets.
If $A$ has $c$ cycle components, alternating the two matching colors gives
$2^c$ ordered decompositions and $2^{c-1}$ unordered decompositions.
Thus only connected graphs have a unique unordered decomposition.
For example, the block diagonal matrix with two $2\times2$ all-ones blocks
has two unordered decompositions.
\end{example}

\begin{example}[Two-alphabet Frobenius series]
\label{regularMatrixTwoAlphabets}
Write $F_{n,2}(X,Y)$ for the characteristic of $V_{n,2}$, normalized by
\[
 F_{n,2}(X,Y)=\frac1{(n!)^2}
 \sum_{\sigma,\tau\in\symS_n}\chi_{n,2}(\sigma,\tau)
 \powerSum_{\operatorname{type}(\sigma)}(X)
 \powerSum_{\operatorname{type}(\tau)}(Y).
\]
The coefficient of $\schurS_\alpha(X)\schurS_\beta(Y)$ is the multiplicity
of the outer tensor product $S^\alpha\boxtimes S^\beta$.
For a connected cycle, its characteristic is
\[
 C_m(X,Y)=\frac1{2m}\sum_{g\in D_m}
 \powerSum_{\operatorname{type}_{\rm row}(g)}(X)
 \powerSum_{\operatorname{type}_{\rm col}(g)}(Y).
\]
The two cycle types come from the actual row and column actions of $g$;
they need not coincide. For even $m$, a reflection fixes two vertices
in one class and none in the other.

Assembling an unordered set of components gives
\[
 \sum_{n\geq0}t^nF_{n,2}(X,Y)
 =\operatorname{Exp}\left[\sum_{m\geq2}t^mC_m(X,Y)\right],
 \qquad F_{0,2}=1.
\]
Here $\operatorname{Exp}[A]=\exp(\sum_{r\geq1}\psi_r(A)/r)$, where
$\psi_r$ replaces $p_i(X),p_i(Y),t$ simultaneously by
$p_{ri}(X),p_{ri}(Y),t^r$. Equivalently, the type-$\lambda$ contribution is
$\prod_{m\geq2}\completeH_{a_m}[C_m]$, the characteristic of the
wreath-product orbit above. This is the usual induction/plethysm
correspondence \cite[Chap.~I, Sec.~8]{Macdonald1995}.
For example,
\[
 C_2=\completeH_2(X)\completeH_2(Y),\qquad
 F_{4,2}=C_4+\completeH_2[C_2].
\]
Burnside's orbit count is the trivial multiplicity:
$\langle F_{n,2},\completeH_n(X)\completeH_n(Y)\rangle=p(n)-p(n-1)$
for $n\geq1$, where $p(n)$ is the partition number.
\end{example}

\begin{example}[Diagonal restriction versus numerical span]
\label{regularMatrixDiagonalAction}
Restricting to $(\sigma,\sigma)$ acts by simultaneous relabeling.
If $F_{n,2}=\sum_{\alpha,\beta}m_{\alpha,\beta}
\schurS_\alpha(X)\schurS_\beta(Y)$, its characteristic becomes
\[
 \sum_{\alpha,\beta}m_{\alpha,\beta}
       (\schurS_\alpha*\schurS_\beta),
\]
where $*$ is the \hyperref[schurKroneckerCoefficients]{Kronecker product}.
Setting $X=Y$ instead gives an ordinary product of degree $2n$, not this
degree-$n$ characteristic.

For $n=3$, every matrix is $J_3-P$ for a permutation matrix $P$.
Independent row/column relabeling gives
$F_{3,2}=\sum_{\lambda\vdash3}\schurS_\lambda(X)\schurS_\lambda(Y)$.
Diagonal relabeling is conjugation on $\symS_3$, whose fixed-point counts
on the types $(1^3),(2,1),(3)$ are $6,2,3$. Thus its characteristic is
\[
 3\schurS_3+\schurS_{21}+\schurS_{111}.
\]
The three trivial summands count the three conjugacy classes.

In contrast, the numerical span of these six matrices has dimension $5$.
Its constant part is spanned by $J_3$, and the zero-row-sum and
zero-column-sum part is $S^{21}\boxtimes S^{21}$, of dimension $4$.
Indeed, $A-\tfrac23J_3$ is nonzero in this irreducible outer tensor product,
so its orbit spans that part. The formal module $V_{3,2}$ has dimension $6$.
\end{example}

\section[representationGln]{Representation theory of $\GL_n$}

We can also study representation theory of infinite groups,
such as the group of invertible $n\times n$-matrices with complex entries.
This is the \defin{general linear group}\label{generalLinearGroup},
$\GL_n(\setC)$.
For a brief introduction to representation
theory of $\GL_n(\setC)$, see for example \cite[Chap.~7, App.~2]{StanleyEC2}.
For more detailed introductory accounts, see
\cite{Kamnitzer2009x} and
\name{David E. Speyer}'s blog post on writing down representations of
$\GL_n$ \cite{Speyer2008GLnRepresentations}.
For broader lecture notes on representation theory, see
\name{Zvi Rosen}'s notes from a course by \name{Mark Haiman}
\cite{Rosen2012RepresentationTheoryNotes} and the notes of
\name{Pavel Etingof} et al. \cite{EtingofEtAl2011RepresentationTheory}.


A \defin{representation}\label{glRepresentation} of $\GL_n(\setC)$ is
a group homomorphism $\phi : \GL_n(\setC) \to \GL_m(\setC)$ for some $m$.
We will only study the cases when $\phi$ is
\defin{homogeneous}\label{homogeneousRepresentation}
and \defin{rational}\label{rationalRepresentation},
meaning that $\phi(\alpha A) = \alpha^k \phi(A)$
for some $k \in \setZ$ and all $\alpha \in \setC \setminus \{0\}$.
The integer $k$ is called the \defin{degree}\label{representationDegree}
of $\phi$, and if $k\geq 0$, we say that $\phi$ is a
\defin{polynomial representation}\label{polynomialRepresentation}.

As a concrete example, $\phi : \GL_n(\setC) \to \GL_1(\setC)$ defined as
$\phi(A) = \det(A)$ is a homogeneous polynomial representation of degree $n$.


Let $\phi$ be a rational representation of $\GL_n(\setC)$.
If $A$ has eigenvalues $x_1,\dotsc,x_n$, there is a Laurent polynomial
\[
\glChar(\phi)(\xvec) = \sum_{\alpha \in \setZ^n} m_\alpha x^\alpha
\]
called the \defin{character}\label{glCharacter} of $\phi$,
such that $\phi(A)$ has eigenvalue $x^\alpha$ with multiplicity $m_\alpha$.
Moreover, if $\phi$ is polynomial, then $\glChar(\phi)(\xvec)$ is a polynomial
in $x_1,\dotsc,x_n$.



\begin{theorem}
Every irreducible homogeneous polynomial representation $\phi$ of $\GL_n(\setC)$
is given as
\[
\glChar(\phi)(\xvec) = \schurS_\lambda(x_1,\dotsc,x_n)
\]
for some partition $\lambda$ with at most $n$ parts, where
$\schurS_\lambda$ is a \hyperref[schurS]{Schur polynomial}.
\end{theorem}

Given two characters $\phi$ and $\varphi$, we can define the tensor product
$\phi \otimes \varphi$.  We then have
\[
\glChar(\phi \otimes \varphi)(\xvec) = \glChar(\phi)(\xvec) \cdot \glChar(\varphi)(\xvec).
\]
In particular, if $\phi$ and $\varphi$ are irreducible, then
the character $\glChar(\phi \otimes \varphi)(\xvec)$ decomposes into irreducible
representations via the \hyperref[schurLittlewoodRichardson]{Littlewood--Richardson rule}.

For dominant weights $\lambda,\mu$ and a Weyl-group element $w$,
\name{Manika Gupta}, \name{K.~N.~Raghavan}, and
\name{Sankaran Viswanath} study the \defin{Kostant--Kumar module}
\label{kostantKumarModule}
\[
 \mathcal K(\lambda,w,\mu)
 =U(\mathfrak g)(v_\lambda\otimes v_{w\mu}).
\]
These modules interpolate between the Cartan component and the full tensor
product.  They give presentations, describe the filtration by Demazure
projections, and interpret the resulting refined tensor-product
multiplicities \cite{GuptaRaghavanViswanath2026x}.


If $\phi : \GL(U) \to \GL(V)$ and $\varphi : \GL(V) \to \GL(W)$ are representations,
then the composition $\varphi \circ \phi$ is a representation.
The character of $\varphi \circ \phi$ is related to the individual characters as
\[
\glChar(\varphi \circ \phi) = \glChar(\varphi)[ \glChar(\phi) ],
\]
where we use the \hyperref[plethysm]{plethysm} operation.


\section[repTheoryMore]{Additional topics}

For the interaction between representation theory and
\hyperref[zeroHeckeAlgebra]{0-Hecke algebras}, see
\cite{HicksMillerBrown2024x}.

\name{N. V. Tsilevich}, \name{A. M. Vershik}, and \name{S. Yuzvinsky}
associate an intrinsic hyperplane arrangement to an arbitrary irreducible
representation of the symmetric group \cite{TsilevichVershikYuzvinsky2020}.

\name{Tom Halverson} and \name{Theodore N. Jacobson} construct irreducible
modules for the partition algebra and many diagram subalgebras using
set-partition tableaux \cite{HalversonJacobson2020}.  Their character
formulas are expressed as nonnegative sums of symmetric-group characters.

\name{Brendan Pawlowski}, \name{Eric Ramos}, and \name{Brendon Rhoades}
study representation stability for spanning configurations of subspaces in a
fixed complex vector space \cite{PawlowskiRamosRhoades2023}.  The spanning
line-configuration case is tied to modules arising in the Delta conjecture.

\name{Victor Reiner} and \name{Mark Shimozono} define
\defin{generalized flagged Schur modules}\label{generalizedFlaggedSchurModule}
and \defin{key modules}\label{keyModule} in their
work on \hyperref[key]{key polynomials} and a flagged
\hyperref[schurLittlewoodRichardson]{Littlewood--Richardson rule}
\cite{ReinerShimozono1995}.  In this sense the key and
\hyperref[schurFlagged]{flagged Schur} families also have a
representation-theoretic model as characters of modules for a Borel subgroup.

In \cite{Szendroi2026x}, the coordinate rings of general Segre embeddings of
products of projective spaces are studied.  Their bigraded Hilbert polynomials
are expressed in terms of major-descent generating functions of words in
multisets.  The paper also discusses quantum cohomology and
Garsia--Stanton-style bases.
