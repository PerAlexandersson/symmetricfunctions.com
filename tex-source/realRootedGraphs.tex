\metadescription{Matching and rook polynomials, chromatic w-polynomials of

\section[realRootedMatching]{Matching and independence polynomials}

This page collects real-rooted graph-polynomial examples.  The main methods
include deletion/interlacing theorems, the
\hyperref[compatiblePolynomials]{compatible-polynomial} viewpoint,
\hyperref[stablePolynomial]{stable polynomials}, and matching-polynomial
technology.

Let $G = (V,E)$ be any graph.
The \defin{matching polynomial}\label{matchingPolynomial} $\mu_G(x)$ is
defined as
\[
\mu_G(x) \coloneqq \sum_{\substack{M \subseteq E(G) \\ \text{$M$ matching}}} x^{|M|}
\]
where $|M|$ is the number of edges in the matching.

In \cite[Thm.~4.2]{HeilmannLieb1972}, it was shown that $\mu_G(x)$
is real-rooted. Moreover, $\mu_G(x)$ interlaces $\mu_{G'}(x)$, where $G'$
is any graph obtained from $G$ by removing a vertex.
This is a very powerful theorem, but it follows
from the more general Chudnovsky--Seymour theorem on independence polynomials below.

A hypergraph generalization (including multigraphs)
is considered by \name{Nima Amini} in \cite{Amini2019x}.


\begin{example*}[Laplacian matching polynomial]
Let $d_G(v)$ denote the degree of $v$ in $G$.
The \defin{Laplacian matching polynomial}\label{laplacianMatchingPolynomial}
is the matching-theoretic Laplacian analogue
\[
 \operatorname{LM}(G,x) \coloneqq
 \sum_{\substack{M \subseteq E(G) \\ \text{$M$ matching}}}
 (-1)^{|M|}
 \prod_{v\in V(G)\setminus V(M)} (x-d_G(v)).
\]
It is real-rooted with nonnegative roots, see \cite{WanWangMohammadian2022}.
One way to see why this belongs with matching polynomials is the subdivision identity
\[
 \mathsf{M}(S(G),x)=x^{|E(G)|-|V(G)|}\operatorname{LM}(G,x^2),
\]
where $S(G)$ is the subdivision graph and
\[
 \mathsf{M}(H,x) \coloneqq
 \sum_{\substack{M \subseteq E(H) \\ \text{$M$ matching}}}
 (-1)^{|M|}x^{|V(H)\setminus V(M)|}
\]
is the usual Heilmann--Lieb normalization of the matching polynomial.

For connected $G$, \name{Jiang-Chao Wan}, \name{Yi Wang}, and
\name{Ali Mohammadian} also show that $0$ is a root of
$\operatorname{LM}(G,x)$ if and only if $G$ is a tree; that the number of
distinct positive roots is at least the length of the longest path in $G$; and
that $\operatorname{LM}(G,x)$ and $\operatorname{LM}(G-e,x)$ have
\hyperref[interlacingPolynomials]{interlacing roots} for each edge $e$ of $G$
\cite{WanWangMohammadian2022}.
\end{example*}


\begin{example*}[Fibonacci tilings via matchings]
Recall that the number of ways to tile the one-row Young diagram $(n)$
with tiles of shape $(1)$ and $(2)$ is the Fibonacci number $F_{n+1}$.
If we then define
\[
 F_0(x) \coloneqq 1,\qquad
 F_1(x) \coloneqq 1,\qquad
 F_n(x) \coloneqq F_{n-1}(x) + x\cdot F_{n-2}(x)
\]
we see that $[x^k] F_n(x)$ is the number of ways to tile $(n)$ using exactly
$k$ tiles of shape $(2)$.

By the Heilmann--Lieb theorem, we can see that for all $n$, $F_n(x)$ is real-rooted,
since we can see that $\mu_G(x) = F_n(x)$ for the path graph with $n$ vertices.
\end{example*}




\begin{example*}[Touchard polynomials (blocks in set partitions)]\label{touchardAsMatching}

The \hyperref[touchardPolynomial]{Touchard polynomials} may be defined via
\[
T_n(x) \coloneqq
\sum_{P \in SP(n)} x^{\mathrm{blocks}(P)}
= \sum_{k=1}^n S(n,k) x^k.
\]
\emph{The matching approach:} \cite[Lem.~5.1]{Godsil2017} states
that the Stirling number $S(n,k)$ is equal to the number
of size $n-k$ matchings in the bipartite graph $G_n$ on vertex set
\[
 \{1,2,\dotsc,n\} \cup \{1',2',\dotsc,n'\}
\]
where $i$ has an edge with $j'$ if and only if $i \lt j$.

For example, the graph
$G_4 = \{ \{1,2'\}, \{1,3'\}, \{1,4'\}, \{2,3'\}, \{2,4'\}, \{3,4'\} \}$
has $7=S(4,4-2)$ matchings of size $2$,
and $6=S(4,4-1)$ matchings of size $1$.
We then have $x^n T_n(x^{-1}) = \mu_{G_n}(x)$ where
$\mu_{G_n}(x)$ is the matching polynomial of $G_n$.
Since removal of vertices gives interlacing,
$\mu_{G_{n-1}}(x) \interl \mu_{G_n}(x)$
from which interlacing for Touchard polynomials follows.


\emph{The rook placement approach:}
The matching polynomial $\mu_{G_n}(x)$ for the bipartite graph above is the
\hyperref[rookPolynomial]{rook polynomial} of the staircase board of size
$n-1$.

\end{example*}


\begin{example*}[r-Lah number polynomials and $r$-Touchard polynomials]

An interpretation of $r$-Lah polynomials and $r$-Stirling numbers as
matching polynomials is given in \cite{NyulRacz2021}.
\end{example*}



\subsection[rookPolynomials]{Rook polynomials}

The \defin{rook polynomial}\label{rookPolynomial} of a board $B$ is defined as
\[
  R_B(z) \coloneqq \sum_k r_k(B) z^k
\]
where $r_k(B)$ is the number of ways to place $k$ non-attacking rooks on the
board.
Here, the board can be any subset of some $n {\times} n$-board.
Note that $R_B(x)$ is exactly a matching polynomial, $\mu_G(x)$, for a
bipartite graph $G$ on vertex set $\{v_1,\dotsc,v_n,v'_1,\dotsc,v'_n\}$
where  $(v_i,v'_j)$ is an edge if and only if $(i,j)$ is a cell of the board $B$.

\name{Albert Nijenhuis} \cite{Nijenhuis1976} proved that the rook polynomials
are always real-rooted.
In fact, Nijenhuis shows a weighted result involving permanents.
\begin{theorem}[Nijenhuis, 1976]
Let $A = (a_{ij})$ be a non-negative $m\times n$-matrix.
Then
\[
 R_{A}(z) \coloneqq \sum_{\rho} (-z)^{\length(\rho)} \prod_{j=1}^{\length(\rho)}
  a(\rho_j)
\]
is real-rooted, and all its zeros are nonnegative.  Here we sum over all
ways $\rho$ to place non-attacking rooks on the $m\times n$-board,
and $a(\rho_j)$ is the value of the matrix at position given by the $j$th rook.
\end{theorem}


The \defin{hit polynomial}\label{hitPolynomial} of a board can also be defined
in terms of the $r_k(B)$.
Let $B$ be a subset of the $n\times n$-board.
The hit polynomial is then defined as
\[
 T_B(z) \coloneqq \sum_{k =0}^n k! r_{n-k}(B)(z-1)^{n-k} = \sum_{j=0}^n h_j(B) z^j.
\]
Here, $h_j(B)$ is the number of ways to place $n$ rooks on the $n \times n$-board,
with exactly $j$ rooks in $B$, see \cite{KaplanskyRiordan1946,Haglund2000}.

A theorem by Haglund--Ono--Wagner \cite{HaglundOnoWagner1999}
states that if $B$ is a Ferrers board (the same as a
\hyperref[youngDiagram]{Young diagram}),
then $T_B(z)$ is real-rooted.


\subsection[realRootedChromaticW]{Chromatic $w$-polynomials}

Let $G$ be a graph on $n\geq1$ vertices.  Its
\defin{chromatic $w$-polynomial} is
\[
 w_G(t)
 \coloneqq
 (1-t)^{n+1}\sum_{m\geq0}\chi_G(m)t^m
 =\Psi_t\bigl(\chrom_G(\xvec)\bigr),
\]
where the last equality follows from the
\hyperref[gesselLengthSpecialization]{fundamental-length specialization}.
A graph is \defin{chordal}\label{chordalGraph} if every cycle of length at
least four has a chord.

\begin{lemma}
Let $f$ be a nonzero polynomial with nonnegative coefficients and only real
zeros.  If $a\geq0$ and $b>\deg f$, then
\[
 g(t)=t(1-t)f'(t)+(a+bt)f(t)
\]
has nonnegative coefficients and only real nonpositive zeros.
\end{lemma}

\begin{proof}
Write $f(t)=\sum_{j=0}^s c_jt^j$, where $s=\deg f$.  Then
\[
 [t^j]g(t)=(a+j)c_j+(b-j+1)c_{j-1},
\]
where coefficients outside the range $0,\dotsc,s$ are zero.  These
coefficients are nonnegative since $a\geq0$ and $b>s$.

First suppose that $s\geq1$ and that $f$ has distinct negative zeros
$\rho_1<\dotsb<\rho_s$.  At these zeros,
\[
 g(\rho_i)=\rho_i(1-\rho_i)f'(\rho_i),
\]
so the signs alternate and give a zero of $g$ between each consecutive pair.
The leading coefficient of $g$ is $(b-s)c_s>0$, which gives another zero to
the left of $\rho_1$.  If $a>0$, then $g(0)=af(0)>0$ and
$g(\rho_s)<0$, giving a zero in $(\rho_s,0)$.  If $a=0$, the remaining zero
is $0$.  These account for all $s+1$ zeros of $g$.

Polynomials with repeated nonpositive zeros are coefficientwise limits of
polynomials with distinct negative zeros and the same positive leading
coefficient.  The leading coefficient of $g$ remains $(b-s)c_s>0$, so the
result follows by taking limits.  If $f$ is constant, then
$g=(a+bt)f$, and the claim follows directly.
\end{proof}

\begin{theorem}[Brenti, \cite[Thms.~4.11 and~4.15]{Brenti1992}]
If $G$ is a nonempty chordal graph, then both
\[
 \Psi_t\bigl(\chrom_G(\xvec)\bigr)=w_G(t)
 \qquad\text{and}\qquad
 \Psi_t\bigl(\omega\chrom_G(\xvec)\bigr)
\]
have only real nonpositive zeros.
\end{theorem}

\begin{proof}
We prove the first assertion by building $G$ in reverse perfect elimination
order.  The one-vertex graph has $w_G(t)=t$.  Suppose that the current graph
$H$ has $k$ vertices and chromatic number $r$, and that the next vertex has
$d$ earlier neighbors.  These neighbors form a clique, so $0\leq d\leq r$,
and the new graph $H'$ satisfies
\[
 \chi_{H'}(m)=(m-d)\chi_H(m).
\]
Write $W(t)=w_H(t)$.  Differentiating its defining generating series gives
\[
 w_{H'}(t)
 =t(1-t)W'(t)+\bigl((k+1+d)t-d\bigr)W(t).
\]

Since $\chi_H(m)=0$ for $0\leq m<r$ and $\chi_H(r)>0$, the zero of $W$ at
$0$ has multiplicity exactly $r$.  The
\hyperref[chromaticQuasisymmetricGesselExpansion]{fundamental-positive
expansion} shows that $W$ has nonnegative coefficients and degree at most
$k$.  Write $W=t^rf$.  By induction, $f$ has only real zeros; it also has
nonnegative coefficients and $\deg f\leq k-r$.  The recurrence becomes
\[
 w_{H'}(t)
 =t^r\left[
 t(1-t)f'(t)
 +\bigl(r-d+(k+1+d-r)t\bigr)f(t)
 \right].
\]
Here $r-d\geq0$, while
\[
 k+1+d-r>\deg f.
\]
The lemma therefore completes the induction.  Nonnegativity of the
coefficients forces all real zeros to be nonpositive.

Finally, the omega-reversal identity on the
\hyperref[gesselLengthSpecialization]{fundamental-length specialization}
gives
\[
 \Psi_t\bigl(\omega\chrom_G(\xvec)\bigr)
 =t^{n+1}w_G(1/t).
\]
Thus its nonzero zeros are the reciprocals of the nonzero zeros of $w_G$;
any remaining zeros are at the origin.
\end{proof}




\subsection[realRootedIndependence]{Independence polynomials for claw-free graphs}

Let $G = (V,E)$ be any graph.
The \defin{independence polynomial}\label{independencePolynomial} $I(G,x)$ is
defined as
\[
I(G,x) \coloneqq \sum_{\substack{A \subseteq V(G) \\ \text{$A$ independent}}} x^{|A|}
\]
where $|A|$ is the number of vertices in the independent set.

A graph is called \defin{claw-free}\label{clawFreeGraph} if no induced
subgraph is isomorphic
to the graph with edges $\{\{1,2\},\{1,3\},\{1,4\}\}$.

If $G$ is any simple graph, its \defin{line graph}\label{lineGraph} $L(G)$ is
also simple.
Moreover, $L(G)$ is claw-free, and $I(L(G),x) = \mu_G(x)$.

\begin{theorem}[Chudnovsky--Seymour 2007, \cite{ChudnovskySeymour2007}]
Suppose $G$ is claw-free. Then $I(G,x)$ is real-rooted.
As a corollary, we get the Heilmann--Lieb theorem on matching polynomials.
\end{theorem}

An alternative proof, closer in spirit to the Heilmann--Lieb proof for
matching polynomials, is due to \name{Ferenc Bencs} \cite{Bencs2014}.
It uses Christoffel--Darboux type identities for independence polynomials;
this route is also useful for proving interlacing statements via Wronskian
inequalities.

\begin{proof*}
This very briefly illustrates the ideas. Note that if $v \in V(G)$ and
$\operatorname{nbh}(v)$ is the closed neighborhood consisting of $v$ and all
vertices adjacent to $v$, then
\[
 I(G,x) = I(G \setminus \{v\} ,x)
 + x \cdot I(G \setminus \operatorname{nbh}(v),x).
\]
Essentially, the above sum is broken down into the cases where $v \notin A$
and where $v \in A$.
Observe that if $G$ is claw-free, then any induced subgraph is also claw-free.
Chudnovsky--Seymour then use the notion of \hyperref[compatiblePolynomials]{compatible polynomials}
to obtain their result.
\end{proof*}

This result was conjectured by \name{Richard Stanley} in \cite{Stanley1998},
a vertex-weighted version is given by \name{Alexander Engström} \cite{Engstrom2007}
and an alternative (non-vertex-deletion-recursive)
proof is given by \name{Bodo Lass} \cite{Lass2012}.
\name{Jonathan Leake} and \name{Nick Ryder}
\cite[Thm.~3.5 and Cor.~3.6]{LeakeRyder2019} formulate the weighted theorem as
\defin{same-phase stability}: specializing each vertex variable to a
nonnegative multiple of one variable gives a real-rooted polynomial.
They show that the multivariate independence polynomial has this property
if and only if the graph is claw-free. This is weaker than
\hyperref[stablePolynomial]{multivariate real stability}.

For general graphs, independence polynomials need not be real-rooted.
Nevertheless, there are useful partial results.

\begin{theorem}[Brown--Dilcher--Nowakowski, \cite{BrownDilcherNowakowski2000}]
For any graph $G$, a zero of $I(G,x)$ of smallest modulus is real.
Moreover, if $G$ is well-covered with independence number $\beta(G)$,
then all zeros of $I(G,x)$ lie in the annulus
\[
  \frac{1}{|V(G)|}\leq |z|\leq \beta(G),
\]
with equality on the boundary only when $G$ is complete.
Finally, every well-covered graph $G$ is an induced subgraph of a
well-covered graph $H$ with $\beta(H)=\beta(G)$ such that $I(H,x)$ has
only simple real zeros.
\end{theorem}

The last statement shows that real-rooted independence polynomials are
plentiful among well-covered graph expansions, even though well-covered graphs
themselves may have non-real zeros. For caterpillar trees, there are also
broad \hyperref[unimodalPolynomial]{unimodality} results for independence
polynomials, see
\cite{BahlsEthridgeSzabo2018}; these are unimodality results rather than
real-rootedness theorems.
\name{Rohan Pandey} studies independence polynomials of generalized Petersen
graphs $\mathrm{GP}(n,k)$ using a transfer-matrix computation
\cite{Pandey2026x}.  The numerical evidence suggests a parity dichotomy:
for even $k$ the roots appear to be real and negative, while for odd $k$
they form complex conjugate patterns.  The resulting
\hyperref[conjPetersenIndependence]{conjecture} is that
$I(\mathrm{GP}(n,k),x)$ is real-rooted if and only if $k$ is even.

\name{Yan-Ting Xie} and \name{Shou-Jun Xu} study the complementary
\defin{dependence polynomial}\label{dependencePolynomial}
\[
  D(G,x) \coloneqq \sum_k d_k(G)x^k,
\]
where $d_k(G)$ is the number of $k$-vertex subsets which are not independent
\cite{XieXu2024x}.  They prove ultra-log-concavity for $D(G,x)$ when $G$ is
$K_2\cup 2K_1$-free or has an independent set of size $|V(G)|-2$, and they
characterize the real-rooted cases.  Up to isolated vertices, the
real-rooted examples are $K_2$, $P_3$, $K_3$, $2K_2$, $P_4$, $C_4$, and two
small exceptional graphs shown in their Fig.~1.

\name{Alexander Engström} proved the following weighted generalization of the Chudnovsky--Seymour theorem:
\begin{theorem*}[A. Engström, \cite[Thm.~2.5]{Engstrom2007}]
Let $G = (V,E)$ be a graph, and $w : V \to \setR$ be vertex weights.
The \defin{weighted independence polynomial}\label{weightedIndependencePolynomial}
$I(G,w,x)$ is defined as
\[
I(G,w,x) \coloneqq \sum_{\substack{A \subseteq V(G) \\ \text{$A$ independent}}} x^{|A|} \prod_{v\in A} w(v).
\]
If $G$ is claw-free and all weights are non-negative, then $I(G,w,x)$ is real-rooted.
\end{theorem*}

Weighted independent sets in powers of paths give the
\hyperref[interlacingRenewalRecurrences]{renewal recurrence family},
whose polynomials also satisfy an oriented interlacing window.


\subsection[realRootedSinkPolynomials]{Sink polynomials of orientations}

A \defin{sink} of an orientation is a vertex with no outgoing edge.  For a
graph $G$, the \defin{acyclic sink polynomial} and the \defin{sink
polynomial}\label{sinkPolynomial} are
\[
 S^a_G(t)\coloneqq\sum_{O\in\mathrm{AO}(G)}t^{\mathrm{sink}(O)},
 \qquad
 S_G(t)\coloneqq\sum_{O}t^{\mathrm{sink}(O)},
\]
where the first sum is over \hyperref[acyclicOrientation]{acyclic
orientations} and the second over all $2^{|E(G)|}$ orientations.  For
example, $S^a_{K_3}(t)=6t$ and $S_{K_3}(t)=2+6t$.  The number of acyclic
orientations is $S^a_G(1)=(-1)^{|V(G)|}\chi_G(-1)$, where $\chi_G$ is the
\hyperref[chromaticPolynomial]{chromatic polynomial}.

\begin{theorem}[Alexandersson--Leite, \cite[Thm.~3.1 and Thm.~4.3]{AlexanderssonLeite2026x}]
If $G$ is chordal and claw-free, then $S^a_G(t)$ has only real nonpositive
zeros.  If $G$ is claw-free, then $S_G(t)$ is real-rooted.
\end{theorem}

The proof shifts $t\mapsto1+t$.  For every graph,
\[
 \frac{S_G(1+t)}{2^{|E(G)|}}
 =I\bigl(G,\{2^{-d_G(v)}\}_v,t\bigr)
\]
is a \hyperref[weightedIndependencePolynomial]{weighted independence
polynomial} \cite[Prop.~4.1]{AlexanderssonLeite2026x}, so
\hyperref[realRootedIndependence]{Engström's theorem} applies.  For a chordal
claw-free graph, $S^a_G(1+t)$ is likewise a positive multiple of a weighted
independence polynomial.  In particular, $S^a_{L(G)}(t)$ is real-rooted
whenever the line graph $L(G)$ is chordal.  For natural unit interval
graphs, the \hyperref[chromaticSinkRealRooted]{ascent-refined acyclic sink
polynomial} is real-rooted as well.

A \defin{local edge order} on $G$ chooses, at every vertex, a total order of
the edges at that vertex.  An edge is \defin{jointly minimal} if it is
minimal at both of its endpoints.  Let $N_G(t)$ count local edge orders by
the number of jointly minimal edges; for example $N_{K_{1,3}}(t)=6t$.

\begin{theorem}[Alexandersson--Leite, \cite[Thms.~3.8 and~3.16]{AlexanderssonLeite2026x}]
For every graph $G$,
\[
 \frac{N_G(1+t)}{\prod_{v}d_G(v)!}
 =\mu_G\Bigl(t;\Bigl\{\frac{1}{d_G(u)d_G(v)}\Bigr\}_{uv\in E(G)}\Bigr),
\]
a weighted \hyperref[matchingPolynomial]{matching polynomial}, so $N_G(t)$ is
real-rooted.  Moreover, $S^a_{L(G)}(t)\leq N_G(t)$ coefficientwise, with
equality if and only if $G$ is a forest.  Hence $S^a_{L(T)}(t)$ is real-rooted
for every forest $T$.
\end{theorem}

Local edge orders are the locally transitive orientations of $L(G)$, with
jointly minimal edges as sinks.  Line graphs alone do not suffice: the line
graph of a $12$-cycle with paths of length two attached at two vertices at
distance four has a non-real-rooted acyclic sink polynomial
\cite[Ex.~3.20]{AlexanderssonLeite2026x}.  Cycles are fine:
$S^a_{C_n}(t)=2\sum_{k\geq1}\binom{n}{2k}t^k$ is real-rooted
\cite[Prop.~3.21]{AlexanderssonLeite2026x}.

Ladders are neither chordal nor claw-free.  Let $\mathrm{Lad}_n$ be the ladder
with $n$ rungs and $S_n(t)=S^a_{\mathrm{Lad}_n}(t)$.  Then $S_1=2t$,
$S_2=12t+2t^2$, and
\[
 S_n(t)=(t+6)S_{n-1}(t)+9(t-1)S_{n-2}(t)\qquad(n\geq3).
\]

\begin{theorem}[Alexandersson, 2026]
\label{ladderSinkPolynomials}
For every $n\geq1$, $S^a_{\mathrm{Lad}_n}(t)$ has a simple zero at the origin
and all other zeros are simple and negative.  For $n\geq2$, after removing
the zero at the origin, the zeros of $S^a_{\mathrm{Lad}_n}$ and
$S^a_{\mathrm{Lad}_{n+1}}$ strictly interlace.
\end{theorem}


\subsection[realRootedTilingPolynomials]{Tiling polynomials}

Let $\mu=(\mu_0,\mu_1,\dotsc,\mu_\ell)$ be a partition.  The \defin{Ferrers
tile} of $\mu$ is the bottom-justified polyomino whose $i$th column has
$\mu_i$ cells.  Fix $1\leq d\leq\mu_0$, and tile the
$(\mu_0+d-1)\times n$ board with unit squares and copies of the tile, so that
the bottom-left cell of each tile lies in one of the lowest $d$ rows.  The
\defin{tiling polynomial}\label{tilingPolynomial} $P_{\mu,d,n}(t)$ counts
these tilings by the number of big tiles.  It is the independence
polynomial of the graph of overlapping tile placements
\cite[Lemma~4]{AhlbergAlexandersson2026}.  Taller boards, on which two tiles
may start in the same column, are treated by the
\hyperref[transferMatrixWeightedTilings]{transfer-matrix method}.

\begin{theorem}[Ahlberg--Alexandersson, \cite[Thm.~6]{AhlbergAlexandersson2026}]
For $\mu=(k,k)$ and $d\leq k$,
\[
 \sum_{n\geq0}P_{\mu,d,n}(t)x^n=\frac{1}{1-x-dtx^2},
\]
so $P_{\mu,d,n}(t)=\sum_j\binom{n-j}{j}d^jt^j$, the matching polynomial of
a path with edge weight $d$.  All $P_{\mu,d,n}(t)$ are real-rooted.
\end{theorem}

A tile is \defin{rigid}\label{rigidTilingPolynomials} if $\mu_\ell=1$ and
$\mu_{\ell-1}\geq d$.  Then consecutive tiles in a block without fault lines
must be $\ell$ columns apart.  The L-shaped tiles $\mu=(h,1)$ are the case
$\ell=1$.

\begin{theorem}[Ahlberg--Alexandersson, \cite[Prop.~2 and Thm.~14]{AhlbergAlexandersson2026}]
For a rigid tile,
\[
 \sum_{n\geq0}P_{\mu,d,n}(t)x^n=\frac{1}{1-x(1+tx^\ell)^d},
 \qquad
 P_{\mu,d,n}(t)=\sum_{j\geq0}\binom{d(n-j\ell)}{j}t^j,
\]
and every $P_{\mu,d,n}(t)$ has only real nonpositive zeros.
\end{theorem}

For example, $P_{(h,1),2,3}(t)=1+4t+t^2$.  The
\hyperref[interlacingRenewalTheorem]{weighted renewal theorem}, which
generalizes the \hyperref[brandenLeiteRenewal]{Brändén--Leite theorem},
strengthens this to $P_{\mu,d,n}\interl P_{\mu,d,n+s}$ whenever
$ds\leq d\ell+1$; in particular consecutive tiling polynomials interlace
(Ahlberg--Alexandersson, 2026).  For L-shaped tiles, refining by the row of
the last tile gives an interlacing sequence
\cite[Thm.~5]{AhlbergAlexandersson2026}.  On a cylinder, where the left and
right boundaries are identified, the tiling polynomial of $\mu=(h,1)$ with
$d=h$ is real-rooted as well \cite[Thm.~21]{AhlbergAlexandersson2026}.


\subsection[realRootsChow]{Chow polynomials}

\name{Petter Brändén} and \name{Lorenzo Vecchi} prove real-rootedness of
\defin{Chow polynomials}\label{chowPolynomialDefinition} arising from
totally nonnegative matrices and posets.  The precise interlacing theorem and
its paving-matroid consequence are stated on the
\hyperref[matroidChowPolynomials]{matroid page} \cite{BrandenVecchi2025x}.
\name{Elena Hoster} and \name{Christian Stump} prove real-rootedness of Chow
and augmented Chow polynomials for finite graded simplicial posets with a top
element whenever the $h$-vector is positive \cite{HosterStump2025x}.  This
includes Cohen--Macaulay simplicial posets and, in particular,
\hyperref[latticeOfFlats]{lattices of flats} of uniform matroids.  They also
show that the Chow polynomial of the dual poset interlaces the augmented
Chow polynomial.

\name{Petter Brändén} and \name{Leonardo Saud Maia Leite} study the
conjecture of \name{Christos A. Athanasiadis} and \name{Afrodite
Kalampogia-Evangelinou} that chain polynomials of geometric lattices have only
real zeros \cite{BrandenLeite2025x}.  They prove it for perfect matroid
designs, Dowling lattices, and a class containing lattices of flats of
\hyperref[pavingMatroid]{paving matroids}, and analyze how the property behaves under several lattice
operations.

The three general real-rootedness conjectures for chain, Chow, and augmented
Chow polynomials are false.  \name[Gergely Berczi]{Gergely B{\'e}rczi},
\name{Basile Coron}, \name{Christopher Eur}, and \name{Shiyue Li}
construct matroids whose \hyperref[chainPolynomial]{chain polynomials},
Chow polynomials, augmented Chow polynomials, and $h$-polynomials of the order
complexes of their proper nonempty flats are not log-concave, and hence not
real-rooted \cite[Thms.~A--C and Cor.~D]{BercziCoronEurLi2026x}.  The
chain-polynomial examples have rank $4$, which is minimal because ranks at
most $3$ are real-rooted.  Their ordinary Chow-polynomial examples have rank
$7$, also minimal because ranks at most $6$ are real-rooted, and may be chosen
simple and $3$-connected.  The augmented Chow and $h$-polynomial
counterexamples instead arise from the paper's direct-sum amplification; no
minimal-rank claim is made for those two invariants.  Open conjectures on
\hyperref[conjRefinedChow]{refined} and \hyperref[conjDualChow]{dual} Chow
polynomials are collected on the open problems page.

\begin{theorem}[Alexandersson,
\cite[Thms.~1.1 and 4.2]{Alexandersson2026ParkingSmirnov}]
\label{noncrossingChowPolynomial}
Let $\operatorname{Smir}_{n,m}$ be the words in $[m]^n$ whose adjacent letters
differ.  Then
\[
 H_{NC_{n+1}}(t)=\frac{1}{n+1}
 \sum_{w\in\operatorname{Smir}_{n,n+1}}t^{\des(w)}
\]
is the Chow polynomial of the noncrossing partition lattice $NC_{n+1}$ and is
real-rooted for every $n\geq1$.  More strongly, for fixed $n$ and $m\geq2$,
the refinements by final letter form an interlacing sequence.
\end{theorem}
The proof uses the same last-letter recurrence and
\hyperref[matrixPreservesInterlacingSequences]{matrix interlacing criterion}
as the word-enumerator argument above.  Since $H_{NC_{n+1}}(t)$ is
palindromic and gamma-positive, its
\hyperref[gammaPolynomial]{gamma-polynomial} is real-rooted with
nonpositive zeros \cite[Cor.~4.4]{Alexandersson2026ParkingSmirnov}.
