\metatitle{Diagonal harmonics}

\metadescription{Definitions and background for diagonal harmonics,
Macdonald eigenoperators, the shuffle theorem, and parking-function formulas.}
\metakeywords{diagonal harmonics, shuffle theorem, Macdonald eigenoperator,
nabla operator, parking functions}


For background, see the canonical sources by \name{Jim Haglund}
\cite{qtCatalanBook}, and \name{François Bergeron} \cite{Bergeron2009}.
There has been substantial development since the writing of these books.
\name{Brendon Rhoades} surveys symmetric-function identities tied to
\hyperref[deltaOperator]{Delta operators} and their representation-theoretic and geometric realizations
\cite{Rhoades2022Delta}.
\name{Adriano Garsia}, \name{Jim Haglund}, and \name{Marino Romero} introduce
several symmetric-function tools for \hyperref[deltaOperator]{Delta operators} in
\cite{GarsiaHaglundRomero2018}.
\name{Michele D'Adderio}, \name{Alessandro Iraci}, and
\name{Anna Vanden Wyngaerd} prove decorated Dyck-path and parallelogram-polyomino
formulas for important Delta-conjecture scalar products
\cite{DAdderioIraciWyngaerd2022}.  For generalized coinvariant algebras and
\hyperref[setPartitions]{ordered set partitions} related to the Delta
conjecture, see
\cite{HaglundRhoadesShimozono2018,Griffin2021}.

\name{Eugenio Cainelli}, \name{Lorenzo Luccioli},
\name{Alessandro Iraci}, and \name{Giovanni Paolini} introduce
\emph{$qt$Bench}, a benchmark of 29 open $q,t$-combinatorics problems and one
solved calibration problem, with finite data and automated checks
\cite{CainelliLuccioliIraciPaolini2026x}.  Several problems concern Theta,
Delta, $\Xi$, super-nabla, or path and parking-function statistics.  The
proposed area--bounce bijection remains under assessment, while negative
coefficients rule out statistic interpretations for three other targets.

\section[diagonalHarmonics]{Introduction}

The \hyperref[coinvariantAlgebra]{ordinary coinvariant algebra} uses one
set of variables; here we use two sets with a simultaneous permutation action.

Let \(X=(x_1,\dotsc,x_n)\) and \(Y=(y_1,\dotsc,y_n)\).
The symmetric group \(\symS_n\) acts diagonally on
\(\setC[X,Y]\) by
\[
\sigma(x_i)=x_{\sigma(i)}, \qquad \sigma(y_i)=y_{\sigma(i)}.
\]
Let \(\setC[X,Y]^{\symS_n}_{+}\) be the vector space of diagonal invariant
polynomials with zero constant term, and let
\[
 I_n \coloneqq \left\langle \setC[X,Y]^{\symS_n}_{+}\right\rangle.
\]
The \defin{diagonal coinvariant ring}\label{diagonalCoinvariantRing} is
\[
 \mathrm{DR}_n \coloneqq \setC[X,Y]/I_n.
\]
Equivalently, \(I_n\) is generated by the polarized power sums
\[
 p_{a,b}(X,Y)\coloneqq \sum_{i=1}^{n}x_i^a y_i^b,
 \qquad a,b\geq 0,\quad a+b\gt 0,
\]
see \cite[Sec.~1.2]{Haiman1994} and \cite[Sec.~2]{qtCatalanBook}.

The dual differential-operator picture is the \defin{space of diagonal
harmonics}\label{diagonalHarmonicsSpace}
\[
\mathcal{DH}_n
\coloneqq
\left\{
 f\in\setC[X,Y]:
 \sum_{i=1}^{n}
 \partial_{x_i}^{a}\partial_{y_i}^{b}f=0
 \text{ for all } a,b\geq0 \text{ with } a+b\gt0
\right\}.
\]
The diagonal action makes both \(\mathrm{DR}_n\) and \(\mathcal{DH}_n\) into
bigraded \hyperref[SnRepresentationTheory]{$\symS_n$-modules}.  Under the
usual apolar pairing, the two modules have the same bigraded Frobenius
characteristic.

\name{Mark Haiman} proved that
\[
 \mathrm{Frob}_{q,t}(\mathcal{DH}_n)=\nabla \elementaryE_n,
 \qquad
 \dim \mathcal{DH}_n=(n+1)^{n-1},
\]
settling conjectures of Garsia and Haiman
\cite{Haiman1994,Haiman2002}.  The
\hyperref[dhShuffleTheorem]{Shuffle theorem} gives a combinatorial expression
for \(\nabla\elementaryE_n\), where \(\nabla\) is the
\hyperref[nablaOperator]{Macdonald eigenoperator defined below}.

\begin{example*}[The case \(n=2\)]
For \(n=2\), the harmonic space has basis
\[
 1,\qquad x_2-x_1,\qquad y_2-y_1.
\]
The constant vector spans the trivial representation, while
\(x_2-x_1\) and \(y_2-y_1\) span two copies of the sign representation in
bidegrees \((1,0)\) and \((0,1)\).  Thus
\[
  \mathrm{Frob}_{q,t}(\mathcal{DH}_2)
  = \schurS_2 + (q+t)\schurS_{11},
\]
which agrees with \(\nabla\elementaryE_2\).
\end{example*}


\subsection[dhDefinitions]{Macdonald eigenoperators}

Given a partition $\mu$, we define
\[
B(\mu) \coloneqq \sum_{\square \in \mu} q^{\arm'(\square)} t^{\leg'(\square)}.
\]
Thus, a cell in row $i$, column $j$ contributes with $q^{j-1} t^{i-1}$.

\begin{example*}[Example \(B(441)\)]
We have
\[
B(\mu) = 1+q+q^2+q^3 + t(1+q+q^2+q^3)+t^2,
\]
as the monomials arise from the following Young diagram:
\begin{figure}
\begin{ytableau}
1 & q & q^2 & q^3 \\
t & tq & q^2t & q^3t \\
t^2
\end{ytableau}
\end{figure}
\end{example*}

Let \(f\) be any symmetric function.  The operators
\defin{$\Delta_f$}\label{deltaOperator} and
\defin{$\Delta'_f$}\label{deltaPrimeOperator} on \(\spaceSym(q,t)\) are defined
diagonally on the \hyperref[macdonaldH]{modified Macdonald basis} by
\[
\Delta_f( \macdonaldH_\mu ) \coloneqq f[ B(\mu) ] \cdot  \macdonaldH_\mu
\text{ and }
\Delta'_f( \macdonaldH_\mu ) \coloneqq f[ B(\mu) - 1]\cdot \macdonaldH_\mu.
\]
Note the use of \hyperref[plethysm]{plethystic substitution}.
For \(\mu\vdash n\), the \defin{$\nabla$ operator}\label{nablaOperator} is
the special case
\(\nabla=\Delta_{\elementaryE_n}\), so
\[
\nabla( \macdonaldH_\mu )
= \elementaryE_n[B(\mu)]\macdonaldH_\mu
= \prod_{(i,j) \in \mu} q^{j-1} t^{i-1}\macdonaldH_\mu.
\]
In other words,
\[
\nabla( \macdonaldH_\mu )
= q^{\partitionN(\mu')} t^{\partitionN(\mu)} \macdonaldH_\mu.
\]

\name{Erik Carlsson} and \name{Anton Mellit} give a combinatorial formula
for the nabla operator \cite{CarlssonMellit2025}.  This supplies another
explicit model for the Macdonald-eigenoperator that governs diagonal
harmonics.

\name{François Bergeron}, \name{Jim Haglund}, \name{Alessandro Iraci}, and
\name{Marino Romero} introduce the \defin{super nabla operator}
\label{superNablaOperator}
\cite{BergeronHaglundIraciRomero2023x}.  It is a generic Macdonald
eigenoperator from which the usual nabla operator, Delta operators, and several
other Macdonald eigenoperators are obtained by specialization.

\subsection[dhThetaNegut]{Theta and Neguț operators}

\defin{Theta operators} are degree-raising operators on symmetric functions that
package decorated refinements of Delta-operator expressions.  In particular,
the operators $\Theta_{\elementaryE_k}$ record decorations on rises and valleys
in several extensions of the Delta conjecture.  The rational shuffle theorem
uses a second family, the \defin{Neguț operators} $D_\gamma$, indexed by
integer sequences $\gamma$; their action on the constant function $1$
produces the corresponding Catalanimal symmetric functions.

\name{Alessandro Iraci} and \name{Marino Romero} give elementary symmetric
function expansions for a large family of Delta and Theta operator expressions
at $t=1$ \cite{IraciRomero2024DeltaTheta}.  Their formulas are indexed by
$\gamma$-parking functions and lattice $\gamma$-parking functions, and they
lead to further $e$-positivity conjectures away from the specialization
$t=1$.

At $t=1$, the power-sum Theta operators on positive-degree symmetric
functions are commuting derivations.  Their action on a single-row modified
Macdonald polynomial is identified with a normalized
\hyperref[singleRowMacdonaldCumulantLLTPositivity]{Macdonald cumulant}, which
in turn has a positive expansion in vertical-strip LLT polynomials
\cite[Thms.~1.2--1.3]{HaglundTewari2026x}.

\name{Michele D'Adderio} and \name{Marino Romero} prove identities for Theta
operators \cite{DAdderioRomero2020x}.  These identities clarify how the Theta
operators interact with Delta operators and \hyperref[plethysm]{plethystic substitutions}.

\name{Michele D'Adderio}, \name{Giovanni Interdonato},
\name{Alessandro Iraci}, and \name{Roberto Pagaria} give an explicit
realization of the Neguț operators $D_\gamma$ inside the Carlsson--Mellit
Dyck path algebra $\mathbb{A}_{q,t}$
\cite[Thms.~4.11 and~4.20]{DAdderioInterdonatoIraciPagaria2026x}.  Their
formula computes $D_\gamma F$ for an arbitrary symmetric function $F$ with a
number of operator steps linear in the length of $\gamma$, and recovers the
\hyperref[rationalShuffle]{rational shuffle theorem} without passing through
the elliptic Hall algebra.

They also extend the Theta operators to an endomorphism of
$\mathbb{A}_{q,t}$ and derive explicit commutation rules with $D_\gamma$
\cite[Thms.~5.1, 5.12, and~5.14]{DAdderioInterdonatoIraciPagaria2026x}.
As an application, they prove the Theta conjecture at $q=1$, giving the
rise-and-valley-decorated Dyck-path formula for
$\Theta_{\elementaryE_k}\Theta_{\elementaryE_\ell}
\nabla \elementaryE_{n-k-\ell}$
\cite[Thm.~6.1]{DAdderioInterdonatoIraciPagaria2026x}.

\subsection[dhOperators]{Relations among the operators}

For \(1\leq k\leq n\), the two Delta operators above are related by
\cite{HaglundRemmelWilson2018}
\[
 \Delta_{\elementaryE_k} \elementaryE_n =
 \Delta'_{\elementaryE_k} \elementaryE_n +
 \Delta'_{\elementaryE_{k-1}} \elementaryE_n.
\]




\section[dhShuffle]{$\mathcal{DH}_n$ and the shuffle theorem}


\subsection[dhShuffleProps]{Properties of $\mathcal{DH}_n$}

\subsubsection[diagonalCoinvariantHilbertFrobenius]{Hilbert and Frobenius series}

Let \(\mathcal{F}_{\mathcal{DH}_n}(q,t)\) be the bigraded
\hyperref[frobeniusCharacteristic]{Frobenius characteristic} of
\(\mathcal{DH}_n\).  By Haiman's theorem,
\[
 \mathcal{F}_{\mathcal{DH}_n}(q,t)=\nabla \elementaryE_n.
\]

The (total) dimension of $\mathcal{DH}_n$ is $(n+1)^{n-1}$.
This is sequence \oeis{A000272} and in particular,
is the number of \hyperref[parkingFunctions]{parking functions} of size $n$.
The dimension can be refined by degree.  The bigraded Hilbert series is
\[
 \mathrm{Hilb}_{\mathcal{DH}_n}(q,t)
 \coloneqq
 \left\langle \mathcal{F}_{\mathcal{DH}_n}(q,t), \completeH_{1^n} \right\rangle.
\]
The Shuffle theorem gives the parking-function formula
\[
\mathrm{Hilb}_{\mathcal{DH}_n}(q,t)
= \sum_{w \in PF(n)} q^{\dinv(w)} t^{\area(w)}.
\]
It is an open problem to give a combinatorial argument why this sum is
symmetric in $q$ and $t$.

It is still a major open problem to find a manifestly
\hyperref[schurPositiveDefinition]{Schur-positive}
combinatorial formula for the Schur expansion of
\(\mathcal{F}_{\mathcal{DH}_n}(q,t)\); see
\cite[Prob.~2.15]{qtCatalanBook} and \cite[Sec.~3]{vanWilligenburg2019}.
A combinatorial Schur expansion for the relevant vertical-strip
\hyperref[LLT]{LLT polynomials} would solve this problem as well.

\subsubsection[diagonalSuperspaceCoinvariants]{Bases and superspace coinvariants}

\name{Yuhan Jiang} constructs an explicit vector-space basis for the
alternating component of the diagonal coinvariant ring in terms of bivariate
Vandermonde determinants indexed by \hyperref[qtCatalan]{Dyck paths}
\cite{Jiang2025x}.  The
bidegrees of these basis elements recover the usual $q,t$-Catalan formula,
and the paper also proposes a related basis for generalized diagonal
coinvariants indexed by $m$-\hyperref[qtCatalan]{Dyck paths}.

\name{Satoshi Murai}, \name{Brendon Rhoades}, and \name{Andy Wilson} prove
the Fields conjectures for the superspace coinvariant ring
\cite{MuraiRhoadesWilson2025x}.  They identify its ungraded
$\symS_n$-module structure with a sign-twisted permutation action on ordered
set partitions, and compute the \hyperref[frobeniusCharacteristic]{bigraded
Frobenius characteristic}.

Three 2026 extensions make this superspace picture more uniform.
\name{Yuhan Jiang} and \name{John Lentfer} compute the bigraded Frobenius
characteristic of the type-$B$ $(0,2)$ fermionic coinvariant ring; every
irreducible type-$B$ character occurs with a single Schur-polynomial
multiplicity, giving a multiplicity-free $\GL_2\times B_n$-module
\cite{JiangLentfer2026x}.  \name{Sutanay Bhattacharya} and
\name{Brendon Rhoades} prove the Sagan--Swanson monomial-basis and
inverse-system conjectures for superspace coinvariants of the wreath products
$\setZ_r\wr\symS_n$ \cite{BhattacharyaRhoades2026x}.  Finally,
\name{Brendon Rhoades} and \name{Andy Wilson} give bigraded Hilbert-series
and inverse-system formulas for superspace coinvariants of
$\GL_n(\mathbb F_q)$, extending them to subgroups containing
$\mathrm{SL}_n(\mathbb F_q)$ \cite{RhoadesWilson2026x}.

For the diagonal superspace coinvariant ring $R_n^{(2,1)}$,
\label{diagonalSuperspaceSignComponent}
\name{Nicolle González}, \name{John Lentfer}, and \name{Hanna Mularczyk}
identify the triply graded sign component with that of
$R_n^{(2,0)}\otimes R_n^{(0,1)}$.  Its Hilbert series is the little Schröder
polynomial, proving the sign-character case of Zabrocki's conjecture
\cite[Thms.~4.20--4.21 and Cors.~4.23--4.26]{GonzalezLentferMularczyk2026x}.

\name{Jesse Kim} constructs a combinatorial basis for the fermionic diagonal
coinvariant ring using noncrossing partitions \cite{Kim2023Fermionic}.  The
result gives a combinatorial model for the full ring as an
$\symS_{n-1}$-module, extending earlier descriptions of its maximal degree
components.
\name{Jesse Kim} and \name{Brendon Rhoades} study \hyperref[setPartitions]{set partitions}, fermions,
and skein relations in this setting \cite{KimRhoades2022}.  Their work gives
presentations and bases for related fermionic modules indexed by set
partitions.

\name{Alessandro Iraci}, \name{Brendon Rhoades}, and \name{Marino Romero}
prove the fermionic Theta coinvariant conjecture in the purely fermionic
setting \cite{IraciRhoadesRomero2023FermionicTheta}.  This confirms the
Theta-operator prediction for the corresponding quadruply graded
$\symS_n$-module after setting the commuting variables to zero.

\name{Nancy Wallace} gives explicit combinatorial formulas for some
irreducible characters of the $\GL_k\times \symS_n$-module of multivariate
diagonal harmonics \cite{Wallace2019x}.  For related generalized coinvariant
algebras, see \name{Brendon Rhoades}, \name{Tianyi Yu}, and
\name{Zehong Zhao} on harmonic bases \cite{RhoadesYuZhao2020},
\name{Maria Gillespie} and \name{Brendon Rhoades} on higher Specht bases
\cite{GillespieRhoades2021}, and \name{Matja{\v{z}} Konvalinka} and
\name{Vasu Tewari} on natural extensions of the parking space
\cite{KonvalinkaTewari2021}.

\subsubsection[diagonalCoinvariantExamplesSpecializations]{Examples and specializations}

\begin{example*}[Example of $\nabla \elementaryE_n$ in the Schur basis]
For $n=1,2,3$ we have the following Schur expansions
\begin{array}{ll}
\toprule
n & \nabla \elementaryE_n \\
\midrule
1 & \schurS_{1} \\
2 & \schurS_{2} + (q+t) \schurS_{11} \\
3 & \schurS_{3}+(q^2+t^2+qt+q+t) \schurS_{21} + (q^3+t^3+q^2 t + qt^2 + qt) \schurS_{111} \\
\bottomrule
\end{array}
\end{example*}

Some special values of the Hilbert series are
\[
\mathrm{Hilb}_{\mathcal{DH}_n}(q,0) = [n]_q!, \qquad
q^{\binom{n}{2}}\mathrm{Hilb}_{\mathcal{DH}_n}(q,q^{-1})
= ([n+1]_q)^{n-1}.
\]
These formulas are part of the specialization picture around Haiman's
$n!$ theorem \cite{Haiman1994,Haiman2002}; the latter formula was originally
conjectured by \name{Richard Stanley}.

Also, by summing over spanning trees of $\{0,1,\dotsc,n\}$,
and weighting them by inversions, we have
\[
\mathrm{Hilb}_{\mathcal{DH}_n}(q,1) = \sum_{T} q^{\inv(T)}.
\]

The \hyperref[qtCatalan]{$qt$-Catalan} numbers appear here by computing the
sign representation:
\[
\left\langle \mathcal{F}_{\mathcal{DH}_n}(q,t), \elementaryE_{n} \right\rangle =
\sum_{ P \in \DP(n) } q^{\dinv(P)} t^{\area(P)} =
\sum_{ P \in \DP(n) } q^{\area(P)} t^{\bounce(P)}.
\]
These identities were proved in \cite[Thm.~I.2]{GarsiaHaglund2000}.



\name{Mark Haiman} proved the following
\hyperref[macdonaldPolynomialDefinition]{Macdonald polynomial} formula,
among many other results \cite[Thm.~3.10]{Haiman2002}.
\begin{theorem}[Haiman (2002)]
Set $T_\mu \coloneqq q^{\partitionN(\mu')}t^{\partitionN(\mu)}$,
$M\coloneqq(1-q)(1-t)$, and
\[
w_\mu \coloneqq \prod_{\square \in \mu}
(q^{\arm(\square)} - t^{\leg(\square)+1})
(t^{\leg(\square)}- q^{\arm(\square)+1} ),
\qquad
\Pi_\mu \coloneqq \prod_{\square \in \mu/(1)} \left(1- q^{\arm'(\square)} t^{\leg'(\square)} \right).
\]
Then, with $\macdonaldH_\mu$ being the \hyperref[macdonaldH]{modified Macdonald polynomials},
\[
\mathcal{F}_{\mathcal{DH}_n}(q,t)
= \sum_{\mu \vdash n}
  \frac{ T_\mu M \Pi_\mu B(\mu)}{ w_\mu } \macdonaldH_\mu .
\]
\end{theorem}





\subsection[dhShuffleTheorem]{The shuffle theorem}

The shuffle conjecture was originally stated in \cite{HaglundHaimanLoehrRemmelUlyanov2005}.
It was later refined to what is known as the
\defin{compositional shuffle conjecture}
\label{compositionalShuffleConjecture}, presented in
\cite{HaglundMorseZabrocki2012}.
Both these conjectures have now been settled.


\begin{theorem}[Carlsson--Mellit \cite{CarlssonMellit2017}]
We have the identity
\[
\nabla \elementaryE_n =
\sum_{\pi \in WPF(n)}
  t^{\area(\pi)}q^{\dinv(\pi)}
  \gessel_{n,\mathrm{ides}(\pi)}(\xvec),
\]
where the sum is taken over word parking functions.  The statistic
\(\mathrm{ides}(\pi)\) records the inverse descent set of the parking
function reading word.  The functions \(\gessel_{n,S}\) are the
\hyperref[gessel]{Gessel fundamental quasisymmetric functions}.
\end{theorem}

Applying the zeta map, as defined in \cite[p.~82]{qtCatalanBook},
the right-hand side can be expressed using
vertical-strip \hyperref[unicellularLLT]{LLT polynomials} as
\[
\nabla \elementaryE_n = \sum_{\avec \in \DP(n)} t^{\bounce(\avec)} \LLT_{\avec,\svec}(\xvec;q)
\]
where $\svec$ marks every corner as strict; see
\cite{AlexanderssonPanova2018} for definitions.


Carlsson and Mellit prove the stronger
\defin{compositional shuffle theorem}\label{compositionalShuffleTheorem}.
\begin{theorem}[Carlsson--Mellit \cite{CarlssonMellit2017}]
Let $C_a$ be the operator defined as
\[
C_a f[X] \coloneqq
[z^a] (-1/q)^{a-1} f[X-(1-q^{-1})/z]
\sum_{m\geq 0} z^m \completeH_m[X].
\]
We then have the identity
\[
\nabla (C_{\alpha_1} C_{\alpha_2}\dotsm C_{\alpha_\ell} 1)
=
\sum_{\substack{\pi \in WPF(n) \\ \mathrm{touch}(\pi)=\alpha }}
  t^{\area(\pi)}q^{\dinv(\pi)}
  \gessel_{n,\mathrm{ides}(\pi)}(\xvec),
\]
where \(\alpha\) specifies the touch composition.
\end{theorem}

For more background and references on the Shuffle theorem, see the survey by
\name{Stephanie van Willigenburg} \cite{vanWilligenburg2019}.

\subsection[dhShuffleTheoremGen]{A generalized shuffle theorem}

A generalization of the Shuffle theorem is given in
\cite{BlasiakHaimanMorsePunSeelinger2021x}.  On the combinatorial side, the
sum ranges over lattice paths under any line, not just a diagonal.

\name{Jonah Blasiak}, \name{Mark Haiman}, \name{Jennifer Morse},
\name{Anna Pun}, and \name{George H. Seelinger} prove a nonsymmetric
shuffle theorem in \cite{BlasiakHaimanMorsePunSeelinger2025Shuffle}.
Their framework uses modified \hyperref[nonSymmetricMacdonaldPolynomial]{nonsymmetric
Macdonald polynomials}, a nonsymmetric nabla operator, and nonsymmetric
analogues of \hyperref[flaggedLLT]{flagged LLT polynomials}, and it supplies a natural setting for atom-positivity
questions around nonsymmetric diagonal harmonics.



\subsection[narayanaNumbers]{$qt$-Narayana numbers}


The $qt$-Narayana numbers are described in
\cite{AvalDAdderioDukesHicksBorgne2014}.
They may be defined as
\[
N_{a,b}(q,t) = \sum_{P \in Polyo(b,a+1-b)} q^{\area(P)}t^{\bounce(P)}
\]
where the sum is taken over certain pairs of
\hyperref[lgvLemma]{nonintersecting paths} confined to
a $b \times (a+1-b)$-rectangle, called
\defin{parallelogram polyominoes}\label{parallelogramPolyomino}.

The authors show in \cite[Thm.~6.1]{AvalDAdderioDukesHicksBorgne2014} that
\[
N_{a,b}(q,t) =
(qt)^{a} \langle \nabla \elementaryE_{a-1}, \completeH_{b-1} \completeH_{a-b} \rangle.
\]
The specialization at \(q=t=1\) can also be seen directly from the Shuffle
theorem.
\begin{proposition}[Narayana coefficient at \(q=t=1\)]
For \(0\leq k\leq n\),
\[
\left.
\langle \nabla \elementaryE_n,\completeH_k\completeH_{n-k}\rangle
\right|_{q=t=1}
=
\frac{1}{n+1}\binom{n+1}{k+1}\binom{n+1}{k}.
\]
\end{proposition}
\begin{proof*}
By the Shuffle theorem and Hall duality, the scalar product is the number of
\hyperref[wordParkingFunctionDefinition]{word parking functions} of size \(n\)
with monomial \(x_1^k x_2^{n-k}\).
With only the labels \(1\) and \(2\), every column has height at most two.
If the supporting Dyck path has \(j\) columns of height two, then the number
of such paths is the Motzkin-triangle number
\[
  \frac{n!}{(n-2j)!j!(j+1)!}.
\]
The \(j\) two-car columns must contain both labels, and among the remaining
\(n-2j\) one-car columns we choose \(k-j\) columns to carry label \(1\).
Thus the desired number is
\[
\sum_{j\geq 0}
\frac{n!}{(n-2j)!j!(j+1)!}\binom{n-2j}{k-j}.
\]
Multiplying by \(n+1\), rewriting the summand, and applying Vandermonde gives
\[
\begin{aligned}
(n+1)\sum_{j\geq 0}
\frac{n!}{(n-2j)!j!(j+1)!}\binom{n-2j}{k-j}
&=
\binom{n+1}{k+1}
\sum_{j\geq 0}\binom{k+1}{j+1}\binom{n-k}{j}  \\
&=
\binom{n+1}{k+1}\binom{n+1}{k}.
\end{aligned}
\]
\end{proof*}

\name{Eugene Gorsky}, \name{Mikhail Mazin}, and \name{Monica Vazirani}
study recursions for rational $q,t$-Catalan numbers
\cite{GorskyMazinVazirani2020}.  \name{François Bergeron} gives a broad list of
open questions for operators connected with rectangular Catalan combinatorics
\cite{Bergeron2017}.

Rational shuffle, Hikita polynomials, square paths, and Delta-type formulas
are collected on the
\hyperref[rationalShuffle]{shuffle and Delta conjectures page}.
