\newcommand{\labeledvee}[4]{%
  \begin{scope}[xshift=#1]
    \coordinate (left) at (-0.8,0);
    \coordinate (right) at (0.8,0);
    \coordinate (top) at (0,1.2);
    \draw[poset edge] (left) -- (top) -- (right);
    \node[poset node,label=below:{$a$}] at (left) {#2};
    \node[poset node,label=below:{$b$}] at (right) {#3};
    \node[poset node,label=above:{$c$}] at (top) {#4};
  \end{scope}%
}

\newcommand{\idealvee}[5]{%

    \node[poset node,minimum size=5.8mm,#2] at (-0.55,0) {};
    \node[poset node,minimum size=5.8mm,#3] at (0.55,0) {};
    \node[poset node,minimum size=5.8mm,#4] at (0,0.82) {};
    \node[below] at (0,-0.38) {#5};
  \end{scope}%
}
