\newcommand{\growthDiagramEntryRow}[2]{%
  \foreach [count=\x from 0] \cellvalue in {#2} {%
    \ifnum\cellvalue=1
      \growthDiagramEntry[
        circle,
        draw=matrixblue,
        fill=matrixblue!35,
        line width=0.6pt,
        minimum size=5mm,
        font=\scriptsize\bfseries,
        text=black
      ]{\x}{#1}{\cellvalue}%
    \else
      \growthDiagramEntry{\x}{#1}{\cellvalue}%
    \fi
  }%
}

\newcommand{\growthDiagramVertexRow}[2]{%
  \foreach [count=\x from 0] \shape in {#2} {%
    \growthDiagramVertex{\x}{#1}{\shape}%
  }%
}
